<%*
/**
 * 03. Create-Note.2'nd
 *
 * 요구사항 반영
 * 부모 페이지
 *  - 1 Line Property(맨 앞 ---) ~ 마지막 목차 라인까지: 빈 줄(공백 개행) 없이 처리
 *  - 마지막 목차 라인 이후: 개행 1줄만 추가 후 이후 내용 유지
 *
 * 자식 페이지
 *  - 1 Line Property(맨 앞 ---) ~ 프로퍼티/내용 분리용 --- 까지: 빈 줄 없이 처리
 *  - 분리용 --- 이후: 개행 1줄 추가하여 이후 작업 편의성 제공
 *
 * 기타 기능
 *  - 현재 문서와 동일 폴더에 하위 문서 생성
 *  - 하위 문서 파일명: "<major>.<minor>. <제목>.md"
 *  - 하위 문서 tags: 폴더 경로 기반 자동 생성
 *  - 부모 outgoinglinks / 자식 backlinks 자동 추가 및 정렬
 */

// ─────────────────────────────────────
// 0. 기본 정보
// ─────────────────────────────────────
const title  = tp.file.title;
const folder = tp.file.folder(true);
const curDir = folder;
const curPathNoExt = `${folder}/${title}`;
const curPath      = `${curPathNoExt}.md`;

if (title === "Untitled") {
	new Notice("❌ 'Create new note from template' 가 아니라 'Insert template' 로 실행해야 합니다.");
	return;
}
if (!/^\d+\./.test(title)) {
	new Notice("❌ 파일명은 '01. 제목' 형식이어야 합니다. (예: '01. Test.md', '01.02. Test.md', '01.02.03. Test.md')");
	return;
}

// 현재 파일의 번호 체계 파싱 (예: "01.02.03. 제목" → "01.02.03.")
const numberPrefixMatch = title.match(/^((?:\d+\.)+)/);
const currentPrefix = numberPrefixMatch ? numberPrefixMatch[1] : "";
// ["01", "02", "03"] 형태로 분리
const prefixParts = currentPrefix.split('.').filter(p => p !== '');
const currentLevel = prefixParts.length;

// 호환성을 위해 major 유지 (첫 번째 숫자)
const major = prefixParts.length > 0 ? prefixParts[0] : "01";

// ─────────────────────────────────────
// 유틸: 자식 문서 frontmatter/절단선 개행 정리
// ─────────────────────────────────────
const normalizeChildFrontmatterSpacing = (content) => {
	if (!content) return content;

	// 파일 최상단 공백 제거
	let updated = content.replace(/^(\s*\r?\n)+(?=---)/, "");
	
	// frontmatter 블록 찾기 (첫 번째 --- ~ 두 번째 ---)
	const fmBlockMatch = updated.match(/^(---[\s\S]*?^---)\s*$/m);
	if (!fmBlockMatch) return updated.replace(/\n{3,}/g, "\n\n");

	// frontmatter 내부 빈 줄 제거
	let frontmatter = fmBlockMatch[1]
		.replace(/\r/g, "")
		.replace(/\n\s*\n+/g, "\n");

	// frontmatter 이후 모든 내용
	let afterFm = updated.slice(fmBlockMatch[0].length);
	
	// frontmatter와 절단선 사이의 모든 공백 제거
	afterFm = afterFm.replace(/^\s+/, "");
	
	// 절단선 찾기
	const dividerMatch = afterFm.match(/^---\s*(?:\r?\n|$)/);
	let body = "";
	
	if (dividerMatch) {
		// 절단선 이후 내용
		body = afterFm.slice(dividerMatch[0].length);
		// 절단선 직후의 모든 공백 제거
		body = body.replace(/^\s+/, "");
		// 본문이 빈 절단선들만 있으면 완전히 제거
		if (/^(?:---\s*(?:\r?\n|$))*\s*$/.test(body)) {
			body = "";
		}
	}

	// 최종 형식: frontmatter + 개행 + 절단선 + 개행 + 본문
	const normalized = `${frontmatter}\n---\n${body}`;
	return normalized;
};

// ─────────────────────────────────────
// 1. 하위 제목 입력
// ─────────────────────────────────────
const childTitleInput = await tp.system.prompt("하위 문서 제목을 입력하세요 (예: ERP)");
if (!childTitleInput || childTitleInput.trim() === "") {
	new Notice("❌ 하위 문서 제목이 없어 중단합니다.");
	return;
}
const childTitleRaw = childTitleInput.trim();

// ─────────────────────────────────────
// 2. 부모 문서 읽기 (frontmatter + body 분리)
// ─────────────────────────────────────
const curTFile = await app.vault.getAbstractFileByPath(curPath);
if (!curTFile) {
	new Notice(`❌ 현재 파일을 찾을 수 없습니다: ${curPath}`);
	return;
}
let originalContent = await app.vault.read(curTFile);

// 2-1) 첫 frontmatter 블록(--- ~ ---)
let fm = "";
let restAfterFm = originalContent;
const fmMatch = originalContent.match(/^---[\s\S]*?^---\s*\r?\n?/m);
if (fmMatch) {
	const fmEndIndex = fmMatch.index + fmMatch[0].length;
	fm = originalContent.slice(0, fmEndIndex);
	restAfterFm = originalContent.slice(fmEndIndex);
}

// 2-2) 두 번째 '---' (내용 시작 구분선)
let headerPart = fm;
let bodyPart   = restAfterFm;
const secondDelimMatch = restAfterFm.match(/^---\s*\r?\n/);
if (secondDelimMatch) {
	headerPart = fm + secondDelimMatch[0];                     // 프로퍼티 + 분리용 ---
	bodyPart   = restAfterFm.slice(secondDelimMatch[0].length); // 분리선 아래부터 본문
}

// ─────────────────────────────────────
// 3. 기존 목차에서 다음 레벨 번호 계산
// ─────────────────────────────────────
const bodyLines = bodyPart.split("\n");

// 현재 prefix로 시작하는 하위 항목 패턴 생성
// 예: currentPrefix가 "01.02."이면 "01.02.XX." 형태의 항목 찾기
// 주의: 제목에 괄호가 포함될 수 있으므로 링크 부분은 .+ (greedy)로 매칭
const escapedPrefix = currentPrefix.replace(/\./g, '\\.');
const childPatternStr = `^\\s*\\[${escapedPrefix}(\\d+)\\. [^\\]]+\\]\\(.+\\)\\s*$`;
const childPattern = new RegExp(childPatternStr);

// 일반 목차 패턴 (정렬용) - 모든 레벨의 번호 체계 지원
const tocRegexParent = /^\s*\[((?:\d+\.)+) [^\]]+\]\([^)]+\)\s*$/;

let maxNextLevel = 0;
for (const line of bodyLines) {
	const m = line.match(childPattern);
	if (!m) continue;
	const num = parseInt(m[1], 10);
	if (!Number.isNaN(num) && num > maxNextLevel) maxNextLevel = num;
}
const nextLevelNum = maxNextLevel + 1;
const nextLevel = String(nextLevelNum).padStart(2, "0");

// ─────────────────────────────────────
// 4. 표시 제목 / 파일명 / TOC 링크 구성
// ─────────────────────────────────────
let childDisplayTitle;
// 이미 번호 체계가 포함된 경우 그대로 사용
if (/^(?:\d+\.)+\s+/.test(childTitleRaw)) {
	childDisplayTitle = childTitleRaw;
} else {
	// 현재 prefix + 다음 레벨 번호 + 제목
	// 예: "01.02." + "01" + ". ERP" → "01.02.01. ERP"
	childDisplayTitle = `${currentPrefix}${nextLevel}. ${childTitleRaw}`;
}

const childFileBase       = childDisplayTitle;
const childFilePath       = `${curDir}/${childFileBase}.md`;
const childFilePathNoExt  = childFilePath.replace(/\.md$/, "");
const childFileUrl        = `${childFileBase}.md`.replace(/ /g, "%20");
const tocLine             = `[${childDisplayTitle}](${childFileUrl})`;

// ─────────────────────────────────────
// 5. 하위 문서 생성 (AB. Template-Note)
// ─────────────────────────────────────
const noteTemplate = await tp.file.find_tfile("AB. Template-Note");
if (!noteTemplate) {
	new Notice("❌ 'AB. Template-Note' 템플릿을 찾을 수 없습니다.");
	return;
}
await tp.file.create_new(noteTemplate, childFileBase, true, curDir);

const childFile = await app.vault.getAbstractFileByPath(childFilePath);
if (!childFile) {
	new Notice("❌ 하위 문서가 생성되지 않았습니다.");
	return;
}

let childContent = await app.vault.read(childFile);

// 5-1) 자식 tags: 폴더 경로 기반 자동 생성
if (childContent.includes("tags")) {
	const parts = curDir.split("/");
	
	// 기존 태그 파싱
	const existingTagsMatch = childContent.match(/tags:\s*\r?\n((?:\s*-\s+[^\n]+\n?)*)/);
	const existingTags = new Set();
	if (existingTagsMatch && existingTagsMatch[1]) {
		const tagLines = existingTagsMatch[1].match(/^\s*-\s+(.+)$/gm);
		if (tagLines) {
			tagLines.forEach(line => {
				const tagMatch = line.match(/^\s*-\s+(.+)$/);
				if (tagMatch) existingTags.add(tagMatch[1].trim());
			});
		}
	}
	
	// 새로운 태그 수집 (Set으로 중복 제거)
	const newTagsSet = new Set();
	for (let i = 0; i < parts.length; i++) {
		const tagName = parts[i].replace(/^\d+\.\s*/, "").toLowerCase().replace(/\s+/g, "_");
		if (!existingTags.has(tagName)) {
			newTagsSet.add(tagName);
		}
	}
	
	// Set을 배열로 변환하여 YAML 형식으로 작성
	let yamlLink = "";
	const newTagsArray = Array.from(newTagsSet);
	for (let i = 0; i < newTagsArray.length; i++) {
		yamlLink += `  - ${newTagsArray[i]}\n`;
	}
	
	if (yamlLink.trim() !== "") {
		childContent = childContent.replace(/tags:\s*\r?\n/, match => `${match}${yamlLink}`);
	}
}

// 5-2) 자식: 파일 최상단 공백 제거 (BOM/빈 줄)
//      → 1 Line Property(---)부터 바로 시작
childContent = childContent.replace(/^(\s*\r?\n)+(?=---)/, "");

// 5-3) 자식: frontmatter/절단선 개행 규칙 적용
childContent = normalizeChildFrontmatterSpacing(childContent);

await app.vault.modify(childFile, childContent);

// ─────────────────────────────────────
// 6. 부모 본문: 목차 정렬 후 재구성
//    - 프로퍼티/분리선(두 번째 ---) ~ 마지막 목차 라인까지 빈 줄 제거
//    - 마지막 목차 이후 개행 1줄만 유지
// ─────────────────────────────────────
// 목차 라인 패턴: [XX.YY.ZZ. 제목](링크) 형태 (모든 레벨 지원)
// 주의: 제목에 괄호가 포함될 수 있으므로 링크 부분은 .+ (greedy)로 매칭
const tocLinePattern = /^\s*\[(?:\d+\.)+\s+[^\]]+\]\(.+\)\s*$/;

let tocLines = [];
let otherLines = [];

for (const line of bodyLines) {
	if (tocLinePattern.test(line)) tocLines.push(line.trim());
	else otherLines.push(line);
}
tocLines.push(tocLine);
tocLines.sort((a, b) => a.localeCompare(b));

// 일단 header + 목차 + 나머지 구성
let newBodyRaw = tocLines.join("\n") + "\n\n" + otherLines.join("\n");
let curContent = headerPart + newBodyRaw;

// 부모: 프로퍼티/분리선 ~ 마지막 목차까지 빈 줄 제거 + 마지막 목차 이후 개행 1줄 유지
{
	let lines = curContent.split("\n");

	// 첫 번째, 두 번째 '---' 찾기
	let firstDelim = -1;
	let secondDelim = -1;
	for (let i = 0; i < lines.length; i++) {
		if (lines[i].trim() === "---") {
			if (firstDelim === -1) firstDelim = i;
			else if (secondDelim === -1) {
				secondDelim = i;
				break;
			}
		}
	}

	// 마지막 목차 라인 찾기 (모든 레벨 지원)
	// 주의: 제목에 괄호가 포함될 수 있으므로 링크 부분은 .+ (greedy)로 매칭
	let lastToc = -1;
	const tocLineRegex = /^\[(?:\d+\.)+\s+[^\]]+\]\(.+\)$/;
	for (let i = 0; i < lines.length; i++) {
		if (tocLineRegex.test(lines[i].trim())) lastToc = i;
	}

	if (firstDelim !== -1 && secondDelim !== -1 && lastToc !== -1) {
		let newLines = [];

		// 0 ~ firstDelim-1: 그대로 유지
		for (let i = 0; i < firstDelim; i++) newLines.push(lines[i]);

		// firstDelim ~ lastToc: 빈 줄 제거 (개행은 라인 끝에만, 공백 라인은 제거)
		for (let i = firstDelim; i <= lastToc; i++) {
			if (lines[i].trim() !== "") newLines.push(lines[i]);
		}

		// 마지막 목차 이후: 개행 1줄 삽입 후 나머지 내용 유지
		newLines.push("");
		let idx = lastToc + 1;
		while (idx < lines.length && lines[idx].trim() === "") idx++;
		for (let i = idx; i < lines.length; i++) newLines.push(lines[i]);

		curContent = newLines.join("\n");
	}

	// 전역적으로 3줄 이상 연속 개행 → 2줄로 축소 (예방 차원)
	curContent = curContent.replace(/\n{3,}/g, "\n\n");
}

// ─────────────────────────────────────
// 7. 부모 outgoinglinks 정렬 유틸
// ─────────────────────────────────────
const sortOutgoingLinksInContent = (content) => {
	if (!content.includes("outgoinglinks")) return content;

	const regex = /(outgoinglinks:\s*\r?\n)([\s\S]*?)(?=^---\s*$)/m;
	const m = content.match(regex);
	if (!m) return content;

	let lines = m[2]
		.split("\n")
		.map(l => l.trimEnd())
		.filter(l => l.trim().startsWith("-"));

	if (lines.length === 0) return content;

	lines.sort((a, b) => a.localeCompare(b));
	const sortedBlock = m[1] + lines.join("\n") + "\n";

	return (
		content.slice(0, m.index) +
		sortedBlock +
		content.slice(m.index + m[0].length)
	);
};

const outgoingItem = `  - "[[${childFilePathNoExt}|${childDisplayTitle}]]"`;

// ─────────────────────────────────────
// 8. 부모 outgoinglinks 갱신
// ─────────────────────────────────────
if (curContent.includes("outgoinglinks")) {
	if (/^outgoinglinks:\s*\[\s*\]\s*$/m.test(curContent)) {
		curContent = curContent.replace(
			/^outgoinglinks:\s*\[\s*\]\s*$/m,
			`outgoinglinks:\n${outgoingItem}`
		);
	} else if (!curContent.includes(outgoingItem)) {
		curContent = curContent.replace(
			/outgoinglinks:\s*\r?\n/,
			match => `${match}${outgoingItem}\n`
		);
	}
	curContent = sortOutgoingLinksInContent(curContent);
} else {
	console.log("❌ 현재 문서에 outgoinglinks 항목이 없습니다. (추가 생성은 생략)");
}

await app.vault.modify(curTFile, curContent);

// ─────────────────────────────────────
// 9. 자식 backlinks 갱신 + 개행 정리 (최종)
// ─────────────────────────────────────
let finalChild = await app.vault.read(childFile);
const backItem = `  - "[[${curPathNoExt}|${title}]]"`;

if (finalChild.includes("backlinks")) {
	if (!finalChild.includes(backItem)) {
		finalChild = finalChild.replace(
			/backlinks:\s*\r?\n/,
			match => `${match}${backItem}\n`
		);
	}
} else {
	console.log("❌ 하위 문서에 backlinks 항목이 없습니다.");
}

// 자식: frontmatter/절단선 개행 규칙 재적용 (최종)
finalChild = normalizeChildFrontmatterSpacing(finalChild);

await app.vault.modify(childFile, finalChild);

new Notice(`✅ 하위 문서 생성 및 목차/링크/개행 정리 완료: ${childDisplayTitle}`);
%>
