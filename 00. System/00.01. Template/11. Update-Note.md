const options = ["Update-Note를 적용하시겠습니까?", "Yes", "No"];
const choice = await tp.system.suggester(options, options);
if (choice === "Yes") {
	new Notice("🟢 Yes를 선택했습니다.");
	// 원하는 로직 실행
} else {
	new Notice("🔴 No를 선택했습니다. 종료합니다.");
	return;
}

const folder = tp.file.folder(true)
const parts = folder.split("/")
const originalTitle = tp.file.title

function extractFolderPrefix(folderName) {
	const match = folderName.match(/^((?:\d+\.)*\d+)\.\s+/);
	if (!match) {
		return null;
	}
	return match[1];
}

function stripFolderPrefix(fileTitle, folderPrefix) {
	const escapedPrefix = folderPrefix.replace(/[.*+?^${}()|[\]\\]/g, "\\$&");
	const prefixedTitleMatch = fileTitle.match(new RegExp(`^${escapedPrefix}\\.(\\d+\\..+)$`));
	if (!prefixedTitleMatch) {
		return fileTitle.trim();
	}
	return prefixedTitleMatch[1].trim();
}

const folderName = parts[parts.length - 1] ?? ""
const folderPrefix = extractFolderPrefix(folderName)
if (!folderPrefix) {
	new Notice("상위 폴더 이름에서 번호를 찾을 수 없습니다.");
	return;
}

const title = stripFolderPrefix(originalTitle, folderPrefix)
if (!title) {
	new Notice("현재 노트 이름에서 제목을 찾을 수 없습니다.");
	return;
}

const curDirPath = `${folder}`
const curFileName = `${folderPrefix}.${title}`
const curFilePath = `${curDirPath}/${curFileName}`

const originContent = tp.file.content

// 기존 파일에서 #tag 형태의 태그 추출
const hashTagRegex = /#([가-힣a-zA-Z0-9_\-]+)/g;
const extractedTags = new Set();
let match;
while ((match = hashTagRegex.exec(originContent)) !== null) {
	extractedTags.add(match[1]);
}
console.log(`📌 추출된 해시태그: ${Array.from(extractedTags).join(', ')}`);

// 원본 콘텐츠에서 첫 줄의 해시태그 제거
let contentLines = originContent.split('\n');
let cleanedContent = contentLines.map((line, index) => {
	if (index === 0) {
		// 첫 줄에서 해시태그 제거
		const cleaned = line.replace(/#([가-힣a-zA-Z0-9_\-]+)/g, '').trim();
		// 첫 줄이 태그만 있어서 비어있게 되면 null 반환 (나중에 필터링)
		return cleaned.length === 0 ? null : cleaned;
	}
	return line;
}).filter(line => line !== null).join('\n');

await tp.file.rename(`${originalTitle}_tmp`)

let templatefile = await tp.file.find_tfile("AB. Template-Note");
if (!templatefile) {
	console.log("❌ AA. Template-Note");
	return;
}
const templateContent = await app.vault.read(templatefile);
console.log(`[Template Content]\n${templateContent}`);
await tp.file.create_new(templatefile, curFileName, true, curDirPath);

let newFile = await app.vault.getAbstractFileByPath(curFilePath + ".md");
if (newFile) {
	let curContent = (await app.vault.read(newFile)).trim() + `\n${cleanedContent}`;
	
	// 기존 태그 파싱
	const existingTagsMatch = curContent.match(/tags:\s*\r?\n((?:\s*-\s+[^\n]+\n?)*)/);
	const existingTags = new Set();
	if (existingTagsMatch && existingTagsMatch[1]) {
		const tagLines = existingTagsMatch[1].match(/^\s*-\s+(.+)$/gm);
		if (tagLines) {
			tagLines.forEach(line => {
				const tagMatch = line.match(/^\s*-\s+(.+)$/);
				if (tagMatch) existingTags.add(tagMatch[1].trim());
			});
		}
	}
	
	// 폴더 경로 기반 태그 생성
	const allTags = new Set(existingTags);
	
	// 폴더 경로 태그 추가
	for (let i = 0; i < parts.length; i++) {
		const folderTag = parts[i].replace(/^\d+\.\s*/, "").toLowerCase().replace(/\s+/g, "_");
		allTags.add(folderTag);
	}
	
	// 추출된 해시태그 추가
	extractedTags.forEach(tag => {
		allTags.add(tag.toLowerCase().replace(/\s+/g, "_"));
	});
	
	// 새로 추가할 태그만 필터링
	const newTags = Array.from(allTags).filter(tag => !existingTags.has(tag)).sort();
	
	if (newTags.length > 0) {
		let yamlLink = "";
		for (let i = 0; i < newTags.length; i++) {
			yamlLinkItem = `  - ${newTags[i]}`;
			if(i != newTags.length - 1)
				yamlLink += `${yamlLinkItem}\n`;
			else
				yamlLink += `${yamlLinkItem}`;
			console.log(`✅ 태그 추가: ${newTags[i]}`);
		}
		
		const updated = curContent.replace(/tags:\s*\r?\n/, match => `${match}${yamlLink}\n`);
		await app.vault.modify(newFile, `${updated}`);
		console.log(`✅ 내용 추가 완료. (${newTags.length}개 신규 태그 추가, 총 ${allTags.size}개)`);
	} else {
		await app.vault.modify(newFile, curContent);
		console.log(`🔁 모든 태그가 이미 존재함 (총 ${allTags.size}개)`);
	}
	curContent = await app.vault.read(newFile);
} else {
	console.log(`❌ 파일이 존재하지 않습니다. (${curFilePath})`);
	return;
}

let mocTemplatefile = await tp.file.find_tfile("AA. Template-MOC");
if (!mocTemplatefile) {
	console.log("❌ AB. Template-Note-MOC 파일을 찾을 수 없습니다.");
	return;
}
const mocTemplateContent = await app.vault.read(mocTemplatefile);
console.log(`[MOC Template Content]\n${mocTemplateContent}`);

const mocDirPath = curDirPath;
const mocFileName = `_MOC_`;
const mocFilePath = `${mocDirPath}/${mocFileName}.md`;
console.log(`[[${mocFilePath}|${title}]]`);

let mocFile = await app.vault.getAbstractFileByPath(mocFilePath);
if (!mocFile) {
	console.log(`Create: ${mocFilePath}`);
	mocFile = await app.vault.create(mocFilePath, mocTemplateContent);
} else {
	console.log(`Find: ${mocFilePath}`);
}

let original = await app.vault.read(mocFile);

// tags 처리
if (original.includes("tags")) {
	// 새로운 태그 수집 (Set으로 중복 제거)
	const mocParts = mocDirPath.split("/")
	const newTagsSet = new Set();
	for (let i = 0; i < mocParts.length; i++) {
		const tagName = mocParts[i].replace(/^\d+\.\s*/, "").toLowerCase().replace(/\s+/g, "_");
		newTagsSet.add(tagName);
	}
	
	// Set을 배열로 변환하여 확인 및 추가
	let yamlLink = "";
	const newTagsArray = Array.from(newTagsSet);
	for (let i = 0; i < newTagsArray.length; i++) {
		yamlLinkItem = `  - ${newTagsArray[i]}`
		if(original.includes(yamlLinkItem)) {
			console.log(`🔁 태그 이미 존재: ${newTagsArray[i]}`);
			continue;
		}
		if(i != newTagsArray.length - 1)
			yamlLink += `${yamlLinkItem}\n`;
		else
			yamlLink += `${yamlLinkItem}`;
		console.log(`✅ 태그 추가: ${newTagsArray[i]}`);
	}
	if (yamlLink || yamlLink.trim() != "") {
		const updated = original.replace(/tags:\s*\r?\n/, match => `${match}${yamlLink}\n`);
		await app.vault.modify(mocFile, updated);
		console.log(`✅ tags 추가됨`);
	} else {
		console.log(`🔁 tags 이미 존재함`);
	}
}

// 최신 MOC 내용 다시 읽기
original = await app.vault.read(mocFile);

// outgoinglinks 처리 + 정렬
if (original.includes("outgoinglinks")) {
	let yamlLink = `  - "[[${curFilePath}|${title}]]"`;
	let updated = original;

	// 1) 현재 노트 링크 추가 (없을 때만)
	if(!original.includes(yamlLink))	{
		updated = original.replace(/outgoinglinks:\s*\r?\n/, match => `${match}${yamlLink}\n`);
		console.log(`✅ Outgoinglinks 추가됨: ${yamlLink}`);
	} else {
		console.log(`🔁 Outgoinglinks 이미 존재함: ${yamlLink}`);
	}

	// 2) outgoinglinks 블록 전체를 사전순 정렬
	const outgoingRegex = /(outgoinglinks:\s*\r?\n)([\s\S]*?)(?=^---\s*$)/m;
	const matchOutgoing = updated.match(outgoingRegex);

	if (matchOutgoing) {
		let lines = matchOutgoing[2]
			.split("\n")
			.map(l => l.trimEnd())
			.filter(l => l.trim().startsWith("-")); // '  - ...' 만 남김

		// 문자열(사전) 순 정렬
		lines.sort((a, b) => a.localeCompare(b));

		const sortedBlock =
			matchOutgoing[1] + (lines.length > 0 ? lines.join("\n") + "\n" : "");

		updated =
			updated.slice(0, matchOutgoing.index) +
			sortedBlock +
			updated.slice(matchOutgoing.index + matchOutgoing[0].length);

		await app.vault.modify(mocFile, updated);
		console.log(`✅ outgoinglinks 사전순 정렬 완료`);
	} else {
		// outgoinglinks 블록 구조를 찾지 못한 경우: 추가만 반영
		await app.vault.modify(mocFile, updated);
		console.log("⚠️ outgoinglinks 정렬 대상 블록을 찾지 못해, 링크 추가만 수행했습니다.");
	}
} else {
	console.log("❌ outgoinglinks를 찾을 수 없습니다.");
}

// Note(backlinks) 처리
let curContent2 = await app.vault.read(newFile);
if (curContent2.includes("backlinks")) {
	let yamlLink = `  - "[[${mocFilePath}|${parts[parts.length - 1]}]]"`;
	if (!curContent2.includes(yamlLink)) {
		const updated = curContent2.replace(/backlinks:\s*\r?\n/, match => `${match}${yamlLink}\n`);
		await app.vault.modify(newFile, updated);
		console.log(`🔗 Backlinks 추가됨: ${yamlLink}`);
	}
} else {
	console.log("❌ backlinks를 찾을 수 없습니다.");
}

// tmp 파일 삭제
if(tp.file.find_tfile(`${curDirPath}/${originalTitle}_tmp.md`)){
	let oldFile = await app.vault.getAbstractFileByPath(`${curDirPath}/${originalTitle}_tmp.md`);
	console.log(`존재`);
	await app.vault.trash(oldFile);
}
