// outgoinglinks 블록 정렬 (content 문자열 기준)
const sortOutgoingLinksInContent = (content) => {
  if (!content.includes("outgoinglinks")) return content;

  const regex = /(outgoinglinks:\s*\r?\n)([\s\S]*?)(?=^---\s*$)/m;
  const m = content.match(regex);
  if (!m) {
    console.log("⚠️ outgoinglinks 정렬 대상 블록을 찾지 못함");
    return content;
  }

  let lines = m[2]
    .split("\n")
    .map(l => l.trimEnd())
    .filter(l => l.trim().startsWith("-"));

  if (lines.length === 0) return content;

  lines.sort((a, b) => a.localeCompare(b));

  const sortedBlock =
    m[1] + lines.join("\n") + "\n";

  const newContent =
    content.slice(0, m.index) +
    sortedBlock +
    content.slice(m.index + m[0].length);

  console.log("✅ outgoinglinks 사전순 정렬 완료");
  return newContent;
};

// tags 블록에서 유효한 디렉토리 경로 태그만 남기고 나머지 제거
const cleanInvalidTagsInContent = (content, validTags) => {
  if (!content.includes("tags")) return content;

  const regex = /(tags:\s*\r?\n)([\s\S]*?)(?=^[a-zA-Z_]+:\s*$|^---\s*$)/m;
  const m = content.match(regex);
  if (!m) {
    console.log("⚠️ tags 정리 대상 블록을 찾지 못함");
    return content;
  }

  let lines = m[2]
    .split("\n")
    .map(l => l.trimEnd())
    .filter(l => l.trim().startsWith("-"));

  const validTagSet = new Set(validTags);
  const beforeCount = lines.length;
  
  // 유효한 태그만 필터링
  lines = lines.filter(line => {
    const tagMatch = line.match(/^\s*-\s+(.+)$/);
    if (tagMatch) {
      const tagName = tagMatch[1].trim();
      return validTagSet.has(tagName);
    }
    return false;
  });

  const afterCount = lines.length;
  const removedCount = beforeCount - afterCount;

  if (removedCount > 0) {
    console.log(`🧹 유효하지 않은 태그 ${removedCount}개 제거됨`);
  }

  const cleanedBlock = m[1] + (lines.length > 0 ? lines.join("\n") + "\n" : "");

  const newContent =
    content.slice(0, m.index) +
    cleanedBlock +
    content.slice(m.index + m[0].length);

  return newContent;
};
