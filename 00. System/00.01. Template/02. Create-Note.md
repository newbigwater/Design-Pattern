<%*
const note = await tp.system.prompt("📖 Please enter a Note!")
if (!note || note.trim() === "") {
	new Notice("❌ note을 입력하지 않아 중단됩니다.");
	return;
}

const folder = tp.file.folder(true)
const parts = folder.split("/")
const title = note.trim()

function extractFolderPrefix(folderName) {
	const match = folderName.match(/^((?:\d+\.)*\d+)\.\s+/);
	if (!match) {
		return null;
	}
	return match[1];
}

const folderName = parts[parts.length - 1] ?? ""
const folderPrefix = extractFolderPrefix(folderName)
if (!folderPrefix) {
	new Notice("상위 폴더 이름에서 번호를 찾을 수 없습니다.");
	return;
}

const curDirPath = `${folder}`
const curFileName = `${folderPrefix}.${title}`
const curFilePath = `${curDirPath}/${curFileName}`

// AB. Template-Note 로 새 노트 생성
let templatefile = await tp.file.find_tfile("AB. Template-Note");
if (!templatefile) {
	console.log("❌ AB. Template-Note");
	return;
}
const templateContent = await app.vault.read(templatefile);
console.log(`[Template Content]\n${templateContent}`);
await tp.file.create_new(templatefile, curFileName, true, curDirPath);

// 새로 만든 노트에 tags 채우기
let newFile = await app.vault.getAbstractFileByPath(curFilePath + ".md");
if (newFile) {
	let curContent = (await app.vault.read(newFile)).trim();
	console.log(`[before]\n${curContent}`);
	
	// 기존 태그 파싱
	const existingTagsMatch = curContent.match(/tags:\s*\r?\n((?:\s*-\s+[^\n]+\n?)*)/);
	const existingTags = new Set();
	if (existingTagsMatch && existingTagsMatch[1]) {
		const tagLines = existingTagsMatch[1].match(/^\s*-\s+(.+)$/gm);
		if (tagLines) {
			tagLines.forEach(line => {
				const tagMatch = line.match(/^\s*-\s+(.+)$/);
				if (tagMatch) existingTags.add(tagMatch[1].trim());
			});
		}
	}
	
	// 새로운 태그 수집 (Set으로 중복 제거)
	const newTagsSet = new Set();
	for (let i = 0; i < parts.length; i++) {
		const tagName = parts[i].replace(/^\d+\.\s*/, "").toLowerCase().replace(/\s+/g, "_");
		if (!existingTags.has(tagName)) {
			newTagsSet.add(tagName);
			console.log(`✅ 태그 추가 예정: ${parts[i]} -> ${tagName}`);
		} else {
			console.log(`🔁 태그 이미 존재: ${tagName}`);
		}
	}
	
	// Set을 배열로 변환하여 YAML 형식으로 작성
	const newTagsArray = Array.from(newTagsSet);
	let yamlLink = "";
	for (let i = 0; i < newTagsArray.length; i++) {
		if (i != newTagsArray.length - 1)
			yamlLink += `  - ${newTagsArray[i]}\n`;
		else
			yamlLink += `  - ${newTagsArray[i]}`;
	}
	
	if (yamlLink.trim() !== "") {
		const updated = curContent.replace(/tags:\s*\r?\n/, match => `${match}${yamlLink}\n`);
		await app.vault.modify(newFile, `${updated}`);
		console.log(`✅ ${newTagsArray.length}개 태그 추가됨`);
	} else {
		console.log(`🔁 모든 태그가 이미 존재함`);
	}
	curContent = await app.vault.read(newFile);
	console.log(`[after]\n${curContent}`);
} else {
	console.log(`❌ 파일이 존재하지 않습니다. (${curFilePath})`);
	return;
}

// 상위 MOC 준비
let mocTemplatefile = await tp.file.find_tfile("AA. Template-MOC");
if (!mocTemplatefile) {
	console.log("❌ AA. Template-MOC 파일을 찾을 수 없습니다.");
	return;
}
const mocTemplateContent = await app.vault.read(mocTemplatefile);
console.log(`[MOC Template Content]\n${mocTemplateContent}`);

const mocDirPath = curDirPath;
const mocFileName = `_MOC_`;
const mocFilePath = `${mocDirPath}/${mocFileName}.md`;
console.log(`[[${mocFilePath}|${title}]]`);

let mocFile = await app.vault.getAbstractFileByPath(mocFilePath);
if (!mocFile) {
	console.log(`Create: ${mocFilePath}`);
	mocFile = await app.vault.create(mocFilePath, mocTemplateContent);
} else {
	console.log(`Find: ${mocFilePath}`);
}

// MOC outgoinglinks 추가 + 정렬
const yamlLink = `  - "[[${curFilePath}|${title}]]"`;
console.log(`yamlLink: ${yamlLink}`);

let original = await app.vault.read(mocFile);
let updatedMoc = original;

// 1) 링크 추가 (없을 때만)
if (!original.includes(yamlLink)) {
	console.log(`original: ${original}`);
	updatedMoc = original.replace(/outgoinglinks:\s*\r?\n/, match => `${match}${yamlLink}\n`);
	console.log(`updated(added): ${updatedMoc}`);
} else {
	console.log(`🔁 이미 존재함: ${yamlLink}`);
}

// 2) outgoinglinks 블록 전체 사전순 정렬
if (updatedMoc.includes("outgoinglinks")) {
	const outgoingRegex = /(outgoinglinks:\s*\r?\n)([\s\S]*?)(?=^---\s*$)/m;
	const matchOutgoing = updatedMoc.match(outgoingRegex);

	if (matchOutgoing) {
		let lines = matchOutgoing[2]
			.split("\n")
			.map(l => l.trimEnd())
			.filter(l => l.trim().startsWith("-")); // '  - ...' 만 유지

		lines.sort((a, b) => a.localeCompare(b));

		const sortedBlock =
			matchOutgoing[1] + (lines.length > 0 ? lines.join("\n") + "\n" : "");

		updatedMoc =
			updatedMoc.slice(0, matchOutgoing.index) +
			sortedBlock +
			updatedMoc.slice(matchOutgoing.index + matchOutgoing[0].length);

		console.log(`✅ outgoinglinks 사전순 정렬 완료`);
	} else {
		console.log("⚠️ outgoinglinks 정렬 대상 블록을 찾지 못해, 링크 추가만 수행했습니다.");
	}
} else {
	console.log("❌ outgoinglinks를 찾을 수 없습니다.");
}

// 정리된 MOC 저장
await app.vault.modify(mocFile, updatedMoc);
console.log(`✅ MOC 갱신 완료`);

// 새 노트에 backlinks 추가
let curContent2 = await app.vault.read(newFile);
let yamlBacklink = `  - "[[${mocFilePath}|${parts[parts.length - 1]}]]"`;
if (!curContent2.includes(yamlBacklink)) {
	if (curContent2.match(/backlinks:\s*\r?\n/)) {
		const updated = curContent2.replace(/backlinks:\s*\r?\n/, match => `${match}${yamlBacklink}\n`);
		await app.vault.modify(newFile, updated);
		console.log(`🔗 Backlinks 추가됨: ${yamlBacklink}`);
	} else {
		console.log("❌ Backlinks 항목 없음");
	}
} else {
	console.log(`🔁 Backlinks 이미 존재함`);
}

// 실행한 템플릿 파일은 휴지통으로 이동
%>
<%* await tp.file.move(`.trash/${tp.file.title}`); %>
